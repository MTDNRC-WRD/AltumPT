    def from_directory(cls, directory, progress_callback=None, exiftool_path=None, allow_uncalibrated=False):
        """
        Create an ImageSet recursively from the files in a directory
        """
        cls.basedir = directory
        matches = []
        for root, dirnames, filenames in os.walk(directory):
            for filename in fnmatch.filter(filenames, '*.tif'):
                matches.append(os.path.join(root, filename))

        images = []

        # Decide which exiftool executable to use
        if exiftool_path is None and os.environ.get('exiftoolpath') is not None:
            exiftool_exe = os.path.normpath(os.environ.get('exiftoolpath'))
        elif exiftool_path is None:
            # Fall back to your hard-coded absolute path
            exiftool_exe = EXIFTOOL_ABS_PATH
        else:
            exiftool_exe = exiftool_path

        # Use explicit executable kwarg so ExifToolHelper does not guess
        with exiftool.ExifToolHelper(executable=exiftool_exe) as exift:
            for i, path in enumerate(matches):
                images.append(image.Image(path, exiftool_obj=exift, allow_uncalibrated=allow_uncalibrated))

                    progress_callback(float(i) / float(len(matches)))
